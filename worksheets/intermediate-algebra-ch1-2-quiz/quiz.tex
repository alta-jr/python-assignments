\quiztitle{Intermediate Algebra}{Chapters 1 \& 2 Quiz}
  {Basic Techniques for Solving Equations \quad$\bullet$\quad Functions Review}

\namedate

\noindent\small
\textbf{Directions.}\ Ten questions, 10 points each. Show all of your work --- a correct
answer with no supporting work earns little credit, and a wrong answer with sound work
earns partial credit. Simplify all answers, and give domains and ranges in interval
notation. No calculators.\par

\parthead{Part I --- Chapter 1}{Basic Techniques for Solving Equations}

\begin{questions}

\item Solve the following inequality. Write your answer in interval notation.
  \[ x - 1 \le 3x + 2 \le 2x + 6 \]
  \work

\item Solve the following system:
  \[
    \begin{aligned}
      4x + 5y &= 43,\\
      9x - 2y &= 57.
    \end{aligned}
  \]
  \work

\item Jeff is 4 times older than his daughter. Five years ago he was 9 times older than
  his daughter. How old is his daughter?
  \work

\newpage

\item Find all ordered pairs $(x,y)$ that satisfy both
  \[ \sqrt{x} + \sqrt{y} = 7 \qquad\text{and}\qquad 3\sqrt{x} - 4\sqrt{y} = -14. \]
  \work

\item Find all solutions to the following system of equations:
  \[
    \begin{aligned}
      8x + y - z &= 46,\\
      3x + 4y - 2z &= 27,\\
      4x - y + z &= 14.
    \end{aligned}
  \]
  \work

\newpage

\parthead{Part II --- Chapter 2}{Functions Review}

\item Find the domain and range of
  \[ S(x) = \frac{12x-9}{6-9x}. \]
  \work

\item Find the domain of
  \[ f(x) = \frac{1}{\sqrt{2x-5}} + \sqrt{9-3x}. \]
  \work

\newpage

\item Suppose that a function $f(x)$ is defined for all real $x$. How can you use the
  graph of $f(x)$ to produce the graph of $y = |f(x)|$? Explain why your method works.
  \work

\item Let $\displaystyle f(x) = \frac{4x}{x+2}$ and $\displaystyle g(x) = \frac{2x}{x+4}$.
  Find $f(g(x))$.
  \work

\item The function $f$ defined by
  \[ f(x) = \frac{cx}{2x+3} \]
  satisfies $f(f(x)) = x$ for all real numbers $x$ except $-3/2$. Find $c$.
  \work

\end{questions}
